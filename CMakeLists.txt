cmake_minimum_required(VERSION 3.20)
project(TazeDecompiler VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/luau/Common/include/Luau/Bytecode.h")
    set(TAZE_DEFAULT_LUAU "${CMAKE_CURRENT_SOURCE_DIR}/luau")
else()
    set(TAZE_DEFAULT_LUAU "${CMAKE_CURRENT_SOURCE_DIR}/luau-master/luau-master")
endif()
set(TAZE_LUAU_DIR "${TAZE_DEFAULT_LUAU}" CACHE PATH "Local, untracked Luau source checkout")
if(NOT EXISTS "${TAZE_LUAU_DIR}/Common/include/Luau/Bytecode.h")
    message(FATAL_ERROR "Set TAZE_LUAU_DIR to a Luau checkout. See README.md.")
endif()

add_library(TazeDecompiler
    src/Bytecode.cpp src/ControlFlow.cpp src/Exports.cpp src/Decompiler.cpp src/Options.cpp src/Batch.cpp
    src/Expressions.cpp src/Symbols.cpp src/Analysis.cpp src/Lowering.cpp
    src/Structure.cpp src/Optimize.cpp src/Render.cpp)
target_include_directories(TazeDecompiler PUBLIC include PRIVATE "${TAZE_LUAU_DIR}/Common/include")
if(MSVC)
    target_compile_options(TazeDecompiler PRIVATE /W4 /EHsc)
else()
    target_compile_options(TazeDecompiler PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
target_link_libraries(TazeDecompiler PRIVATE Threads::Threads)
add_executable(taze src/Main.cpp src/WebSocketServer.cpp src/BatchFiles.cpp)
target_link_libraries(taze PRIVATE TazeDecompiler)
add_executable(taze-bench tools/Benchmark.cpp)
target_link_libraries(taze-bench PRIVATE TazeDecompiler)
add_executable(taze-batch-bench tools/BatchBenchmark.cpp)
target_link_libraries(taze-batch-bench PRIVATE TazeDecompiler)
if(WIN32)
    target_link_libraries(taze PRIVATE ws2_32 bcrypt)
endif()

option(TAZE_BUILD_TESTS "Build compiler/VM round-trip tests and fixture compiler" ON)
if(TAZE_BUILD_TESTS)
    set(LUAU_BUILD_CLI OFF CACHE BOOL "" FORCE)
    set(LUAU_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory("${TAZE_LUAU_DIR}" "luau" EXCLUDE_FROM_ALL)
    enable_testing()
    add_executable(taze-tests tests/Tests.cpp)
    target_link_libraries(taze-tests PRIVATE TazeDecompiler Luau.Compiler Luau.VM)
    target_include_directories(taze-tests PRIVATE "${TAZE_LUAU_DIR}/Common/include")
    target_compile_definitions(taze-tests PRIVATE TAZE_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME RoundTrip COMMAND taze-tests)
    add_executable(taze-pipeline-tests tests/PipelineTests.cpp)
    target_link_libraries(taze-pipeline-tests PRIVATE TazeDecompiler Luau.Compiler Luau.VM)
    add_test(NAME Pipeline COMMAND taze-pipeline-tests)
    add_executable(taze-control-flow-tests tests/ControlFlowStress.cpp)
    target_link_libraries(taze-control-flow-tests PRIVATE TazeDecompiler Luau.Compiler Luau.VM)
    add_test(NAME ControlFlowStress COMMAND taze-control-flow-tests)
    set_tests_properties(ControlFlowStress PROPERTIES TIMEOUT 120)
    add_executable(taze-inline-recovery-tests tests/InlineRecoveryTests.cpp)
    target_link_libraries(taze-inline-recovery-tests PRIVATE TazeDecompiler Luau.Compiler Luau.VM)
    add_test(NAME InlineRecovery COMMAND taze-inline-recovery-tests)
    set_tests_properties(InlineRecovery PROPERTIES TIMEOUT 30)
    add_executable(taze-compile tools/Compile.cpp)
    target_link_libraries(taze-compile PRIVATE Luau.Compiler Luau.VM)
    if(WIN32)
        find_program(TAZE_POWERSHELL NAMES pwsh powershell)
        if(TAZE_POWERSHELL)
            add_test(NAME WebSocketBridge COMMAND "${TAZE_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
                -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/WebSocket.ps1"
                -Executable $<TARGET_FILE:taze> -Compiler $<TARGET_FILE:taze-compile>
                -WorkDirectory "${CMAKE_CURRENT_BINARY_DIR}/WebSocketTests")
            set_tests_properties(WebSocketBridge PROPERTIES TIMEOUT 60)
            add_test(NAME BatchCLI COMMAND "${TAZE_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass
                -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/Batch.ps1"
                -Executable $<TARGET_FILE:taze> -Compiler $<TARGET_FILE:taze-compile>
                -WorkDirectory "${CMAKE_CURRENT_BINARY_DIR}/BatchTests")
            set_tests_properties(BatchCLI PROPERTIES TIMEOUT 60)
        endif()
    endif()
endif()
